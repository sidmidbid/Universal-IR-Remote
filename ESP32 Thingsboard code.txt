#include <WiFi.h>
#include <PubSubClient.h>
#include <ArduinoJson.h>

// ================= UART FOR STM32 =================
#define RXD2 16
#define TXD2 17

// ================= WIFI =================
const char* ssid = "YOUR_WIFI_SSID"; 
const char* password = "YOUR_WIFI_PASSWORD"; 

// ================= THINGSBOARD =================
const char* mqtt_server = "thingsboard.cloud"; 
const char* token = "YOUR_THINGSBOARD_TOKEN"; 

WiFiClient espClient;
PubSubClient client(espClient);

// ================= RPC CALLBACK =================
void callback(char* topic, byte* payload, unsigned int length) {
  StaticJsonDocument<256> doc;
  if (deserializeJson(doc, payload, length)) {
    Serial.println("Failed to parse JSON");
    return;
  }

  String method = doc["method"];
  
  // 1. Process the incoming command from ThingsBoard
  if (method == "powerButton") {
    Serial.println("Dashboard power button pressed. Forwarding to STM32...");
    
    // Send a command string to the STM32 over Serial2
    Serial2.println("SEND_POWER"); 
  }
  // You can add more else-if statements here for other dashboard buttons later
  // else if (method == "volumeUp") { Serial2.println("SEND_VOL_UP"); }

  // 2. Extract the Request ID from the end of the incoming topic string
  String topicStr = String(topic);
  int reqIdIndex = topicStr.lastIndexOf('/') + 1;
  String requestId = topicStr.substring(reqIdIndex);
  
  // 3. Construct the response topic and publish an empty JSON acknowledgement
  String responseTopic = "v1/devices/me/rpc/response/" + requestId;
  client.publish(responseTopic.c_str(), "{}"); 
}

void reconnect() {
  while (!client.connected()) {
    Serial.print("Connecting to ThingsBoard...");
    if (client.connect("ESP32_Universal_Remote", token, NULL)) {
      Serial.println(" [DONE]");
      // Subscribe to RPC commands from dashboard
      client.subscribe("v1/devices/me/rpc/request/+");
    } else {
      Serial.print(" [FAILED] rc=");
      Serial.print(client.state());
      Serial.println(" retrying in 2 seconds...");
      delay(2000);
    }
  }
}

void setup() {
  // Debug output to PC via USB-C
  Serial.begin(115200);
  
  // Hardware UART connection to STM32
  Serial2.begin(115200, SERIAL_8N1, RXD2, TXD2);
  
  Serial.println("\nInitializing Universal Remote Bridge...");

  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi Connected!");

  client.setServer(mqtt_server, 1883);
  client.setCallback(callback);
}

void loop() {
  if (!client.connected()) {
    reconnect();
  }
  client.loop();

  // Listen for incoming status messages from the STM32
  if (Serial2.available()) {
    String stm32Message = Serial2.readStringUntil('\n');
    stm32Message.trim();
    
    Serial.print("STM32 says: ");
    Serial.println(stm32Message);
    
    // Optional: Forward STM32 status back to ThingsBoard telemetry
    // String telemetryPayload = "{\"status\":\"" + stm32Message + "\"}";
    // client.publish("v1/devices/me/telemetry", telemetryPayload.c_str());
  }
}
